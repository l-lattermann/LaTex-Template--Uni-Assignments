% =============================================================================
% IU UNIVERSITY PAPER TEMPLATE
%
% Preconfigured for IU written assignments and project reports:
% Arial 11 pt, DIN A4 with 2 cm margins, 1.5 line spacing, justified text,
% APA references via biblatex + biber, three heading levels, centred page
% numbers in the footer.
%
% USAGE
%   1. Copy this file and rename it to your submission's filename.
%   2. Fill in the <PLACEHOLDERS> on the title page.
%   3. Put your chapters in sections/ and \input them where marked.
%   4. Put your sources in references.bib.
%   5. Build with:  latexmk -pdf <yourfile>.tex
%      (.latexmkrc sends intermediates to build/; latexmk -C cleans up.)
%
% Note that sections/, references.bib and every *.tex other than this file are
% gitignored on purpose - this repository holds the template, not the papers.
% =============================================================================

\documentclass[11pt,a4paper]{article}

% =================================================
% Font: Arial 11 pt or comparable sans-serif
% =================================================
\usepackage[T1]{fontenc}
\usepackage[scaled]{helvet}
\renewcommand{\familydefault}{\sfdefault}

% =================================================
% Page Layout: DIN A4, single-sided, 2 cm margins
% =================================================
\usepackage[
  a4paper,
  top=2cm,
  bottom=2cm,
  left=2cm,
  right=2cm,
  footskip=1cm
]{geometry}

% =================================================
% Language, Hyphenation, Justification
% American English, because APA is an American standard and
% biblatex-apa expects the american language mapping below.
% =================================================
\usepackage[american]{babel}
\usepackage{microtype}
\usepackage{ragged2e}
\justifying
\hyphenpenalty=1000
\tolerance=2000
\emergencystretch=2em

% =================================================
% Line and Paragraph Spacing
% NOTE: 1.5 spacing plus a 6 pt paragraph gap was once criticized as
% "too much" in IU grading feedback; 4 pt is the conservative compromise.
% =================================================
\usepackage{setspace}
\onehalfspacing
\setlength{\parindent}{0pt}
\setlength{\parskip}{4pt}

% =================================================
% Packages
% =================================================
\usepackage{graphicx}
\usepackage{float}   % provides [H]: pin a table exactly where it is written,
                     % instead of letting LaTeX float it past the section that
                     % introduces it.
\usepackage{booktabs}
\usepackage{array}
\usepackage{longtable}
\usepackage[table]{xcolor}
\usepackage{enumitem}
\usepackage{hyperref}
\usepackage{caption}
\usepackage{titlesec}
\usepackage{tocloft}
\usepackage{fancyhdr}
\usepackage{csquotes}

% =================================================
% APA Citation Setup
% Requires: biber
% =================================================
\usepackage[
  backend=biber,
  style=apa,
  sorting=nyt
]{biblatex}
\DeclareLanguageMapping{american}{american-apa}
\addbibresource{references.bib}
\setlength{\bibhang}{1.27cm}

% Reminder: IU expects a locator on every citation, e.g.
%   \parencite[p.~12]{key}
%   \textcite[pp.~12--14]{key}
%   \parencite[Section~4.4.4]{key}   % non-numeric postnotes print verbatim

% =================================================
% Captions and Tables
% =================================================
\captionsetup{
  font=small,
  labelfont=bf,
  justification=justified,
  singlelinecheck=false
}

\renewcommand{\arraystretch}{1.1}
\setlength{\tabcolsep}{8pt}
\newcommand{\tablefontsize}{\fontsize{9}{11}\selectfont}

% =================================================
% Heading Depth: max. three levels
% =================================================
\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{3}

% =================================================
% Headings
% =================================================
\titleformat{\section}
  {\normalfont\fontsize{15}{17}\bfseries}
  {\thesection}{0.75em}{}

\titleformat{\subsection}
  {\normalfont\fontsize{13}{15}\bfseries}
  {\thesubsection}{0.75em}{}

\titleformat{\subsubsection}
  {\normalfont\fontsize{11}{13}\bfseries}
  {\thesubsubsection}{0.75em}{}

\titlespacing*{\section}{0pt}{8pt}{4pt}
\titlespacing*{\subsection}{0pt}{6pt}{3pt}
\titlespacing*{\subsubsection}{0pt}{4pt}{2pt}

% =================================================
% Unnumbered section style
% =================================================
\titleformat{name=\section,numberless}
  {\normalfont\fontsize{16}{19}\bfseries}
  {}{0pt}{}

% =================================================
% Footnotes: Arial/sans-serif 10 pt
% =================================================
\renewcommand{\footnotesize}{\fontsize{10}{12}\selectfont}

% =================================================
% Lists
% =================================================
\setlist{
  itemsep=2pt,
  topsep=6pt,
  partopsep=0pt
}

% =================================================
% Table of Contents
% Only level 1 bold. The Contents and List of Tables titles are forced to the
% same size as an unnumbered \section*, so all front-matter headings match.
% =================================================
\renewcommand{\cfttoctitlefont}{\normalfont\fontsize{16}{19}\bfseries}
\renewcommand{\cftlottitlefont}{\normalfont\fontsize{16}{19}\bfseries}

\renewcommand{\cftsecfont}{\bfseries}
\renewcommand{\cftsecpagefont}{\bfseries}
\renewcommand{\cftsubsecfont}{\normalfont}
\renewcommand{\cftsubsecpagefont}{\normalfont}
\renewcommand{\cftsubsubsecfont}{\normalfont}
\renewcommand{\cftsubsubsecpagefont}{\normalfont}

\renewcommand{\cftdotsep}{1.5}
\renewcommand{\cftsecleader}{\cftdotfill{\cftdotsep}}
\renewcommand{\cftsubsecleader}{\cftdotfill{\cftdotsep}}
\renewcommand{\cftsubsubsecleader}{\cftdotfill{\cftdotsep}}

% =================================================
% Footer: centered page number, no header
% =================================================
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% =================================================
% Hyperlinks: black throughout, so the printed paper stays neutral
% =================================================
\hypersetup{
  colorlinks=true,
  linkcolor=black,
  citecolor=black,
  urlcolor=black,
  pdfborder={0 0 0}
}

% =================================================
% Document
% =================================================
\begin{document}

% =================================================

\begin{titlepage}

  \centering

  \vspace{3cm}

  \includegraphics[height=2.5cm]{images/doc_images/iu-logo.jpg}

  % The university name is deliberately NOT repeated as a text line here: the
  % logo already reads "IU INTERNATIONAL UNIVERSITY OF APPLIED SCIENCES". It
  % still appears as machine-readable text in the details block below, which is
  % what matters for the formal requirement and for text extraction.
  \vspace{0.5cm}
  {\large Degree Programme: <PROGRAMME>\par}

  \vfill

  {\fontsize{8pt}{10pt}\selectfont
  \begin{center}
  \begin{tabular}{@{}p{4cm}p{7cm}}
  \textbf{University:} & IU, International University of Applied Sciences \\
  \textbf{Course/Module:} & <MODULE CODE> -- <MODULE NAME> \\
  \textbf{Assignment:} & <ASSIGNMENT> \\
  \textbf{Name:} & <NAME> \\
  \textbf{Matriculation Number:} & <MATRICULATION NUMBER> \\
  \textbf{Tutor:} & <TUTOR> \\

  \end{tabular}
  \end{center}
  }

\end{titlepage}
