% =============================================================================
% IU UNIVERSITY PAPER TEMPLATE
%
% Preconfigured for IU written assignments and project reports, following the
% IU formatting guidelines (Appendix A): Arial 11 pt, DIN A4 with 2 cm margins,
% 1.5 line spacing, justified text, APA references via biblatex + biber, three
% heading levels at 16/14/11 pt, centred page numbers in the footer.
%
% USAGE
%   1. Copy this file and rename it to the submission file name, e.g.
%      YYYYMMDD_firstname_surname_matriculationnumber_coursecode.tex
%   2. Fill in the <PLACEHOLDERS> on the title page.
%   3. Put the chapters in sections/ and \input them where marked.
%   4. Put the sources in references.bib.
%   5. Build with:  latexmk -pdf <yourfile>.tex
%      (.latexmkrc sends intermediates to build/; latexmk -C cleans up.)
%   6. Switch \reviewtrue to \reviewfalse before submitting.
%
% sections/, references.bib and every *.tex other than this file are
% gitignored on purpose: this repository holds the template, not the papers.
% =============================================================================

% =================================================
% REVIEW MARKS
% \rev{text}         highlights a span in yellow, e.g. a number not yet measured
% \revn{text}{note}  highlight plus a visible bracketed note
% \revcite{key}      a citation still to be checked; stays when review is off
% \revsug{text}      a suggested reformulation; disappears when review is off
% \revbad{note}      red mark for a statement no source supports
% SET \reviewfalse BEFORE SUBMITTING: it turns every mark off in one place and
% leaves the text itself untouched.
% =================================================
\usepackage{soul}
\newif\ifreview
\reviewtrue                       % <-- change to \reviewfalse for the final PDF
\sethlcolor{yellow}
\newcommand{\rev}[1]{\ifreview\hl{#1}\else#1\fi}
\newcommand{\revn}[2]{\ifreview\hl{#1}\,\textcolor{red}{\footnotesize[#2]}\else#1\fi}
\newcommand{\revcite}[1]{\ifreview\hl{\mbox{\parencite{#1}}}\else\parencite{#1}\fi}
\newcommand{\revsug}[1]{\ifreview\hl{[Suggestion: #1]}\else\fi}
\newcommand{\revbad}[1]{\ifreview{\sethlcolor{red!40}\hl{[! #1]}\sethlcolor{yellow}}\else\fi}

% \anchor{id}    a jump destination with no width and no height
% \gl{key}{text} links a term in the body to its glossary or abbreviation entry
\newcommand{\anchor}[1]{\raisebox{0pt}[0pt][0pt]{\hypertarget{#1}{}}}
\newcommand{\gl}[2]{\hyperlink{gl:#1}{#2}}

%   \srcnote{Own representation.}

\newcommand{\srcline}[1]{\emph{Source: #1}}
\newcommand{\srcnote}[2][\linewidth]{%
  {\setlength{\parskip}{0pt}\par\nobreak\vspace{6pt}%
   \noindent\parbox{#1}{\raggedright\itshape\fontsize{10}{12}\selectfont Source: #2}\par}}
